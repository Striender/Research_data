# BASIC USAGE

# Run ALL replacement policies (1-14) sequentially
# Default behavior
./run_all_combination_ai_ml.sh 450

# Same as above but with fewer available cores
./run_all_combination_ai_ml.sh 128


###############################################################################
# SINGLE REPLACEMENT POLICY
###############################################################################

./run_all_combination_ai_ml.sh 450 1

# Run only RP 5 (lru/ship)
./run_all_combination_ai_ml.sh 450 5

# Run only RP 14 (srrip/mockingjay)
./run_all_combination_ai_ml.sh 450 14


###############################################################################
# SELECTED REPLACEMENT POLICIES (SEQUENTIAL)
###############################################################################

# Run RP 1, 3, and 5 sequentially
./run_all_combination_ai_ml.sh 450 rp:1,3,5

./run_all_combination_ai_ml.sh 450 rp:1,2,3,4,5,6,7

# Run all SRRIP-based policies sequentially
./run_all_combination_ai_ml.sh 450 rp:8,9,10,11,12,13,14

###############################################################################
